BLOCK B — docker-compose.yml (repo root)
--------------------------------------------------------------------------------

cat > ~/Audeint-AI/Audient-AI/docker-compose.yml << 'DCYML'
# Audient AI — Docker Compose (Ubuntu / AWS EC2)
#
# Prerequisites:
#   - backend/.env exists (DATABASE_URL, keys, JWT, etc.)
#   - FRONTEND_URL and APP_URL in backend/.env = public URL users open in browser, e.g.
#       http://13.210.144.218
#     (same host as port 80 below — nginx serves SPA + proxies API)
#
# Commands:
#   docker compose build
#   docker compose up -d
#   docker compose logs -f
#
# AWS: security group must allow inbound TCP 80 (and 443 if you add TLS later).

services:
  redis:
    image: redis:7-alpine
    restart: unless-stopped

  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    restart: unless-stopped
    env_file:
      - ./backend/.env
    environment:
      # Overrides empty .env — enables Limiter + Celery queue (worker service below)
      REDIS_URL: redis://redis:6379/0
    volumes:
      - backend_temp:/app/temp
    expose:
      - "5000"
    depends_on:
      - redis

  celery-worker:
    build:
      context: ./backend
      dockerfile: Dockerfile
    restart: unless-stopped
    command: celery -A celery_app.celery worker --loglevel=info
    env_file:
      - ./backend/.env
    environment:
      REDIS_URL: redis://redis:6379/0
    volumes:
      - backend_temp:/app/temp
    depends_on:
      - redis

  web:
    build:
      context: ./frontend
      dockerfile: Dockerfile
      args:
        VITE_API_URL: ""
    restart: unless-stopped
    ports:
      - "80:80"
    depends_on:
      - backend

volumes:
  backend_temp:
DCYML

cat > ~/Audeint-AI/Audient-AI/backend/Dockerfile << 'BDOCKER'
# Audient AI — Flask + Flask-SocketIO (threading mode)
FROM python:3.12-slim-bookworm

RUN apt-get update && apt-get install -y --no-install-recommends \
    ffmpeg \
    libsndfile1 \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir --upgrade pip \
    && pip install --no-cache-dir -r requirements.txt

COPY . .

ENV PYTHONUNBUFFERED=1

EXPOSE 5000

CMD ["python", "app.py"]
BDOCKER

BLOCK E — frontend/Dockerfile

cat > ~/Audeint-AI/Audient-AI/frontend/Dockerfile << 'FDOCKER'
# Build static site, serve with nginx (proxies /api + /socket.io to backend)
FROM node:22-alpine AS build
WORKDIR /app
COPY package.json package-lock.json ./
RUN npm ci
COPY . .
# Empty = same-origin; nginx proxies /api and /socket.io to backend
ARG VITE_API_URL=
ENV VITE_API_URL=$VITE_API_URL
RUN npm run build

FROM nginx:alpine
COPY --from=build /app/dist /usr/share/nginx/html
COPY nginx.docker.conf /etc/nginx/conf.d/default.conf
EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]
FDOCKER


--------------------------------------------------------------------------------
BLOCK F — frontend/nginx.docker.conf
--------------------------------------------------------------------------------

cat > ~/Audeint-AI/Audient-AI/frontend/nginx.docker.conf << 'NGXCONF'
server {
    listen 80;
    server_name _;
    root /usr/share/nginx/html;
    client_max_body_size 50m;

    location / {
        try_files $uri $uri/ /index.html;
    }

    location /api {
        proxy_pass http://backend:5000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 300s;
    }

    location /socket.io {
        proxy_pass http://backend:5000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_read_timeout 86400;
    }
}
NGXCONF

#!/usr/bin/env bash
# Build and start Audient AI stack (from repo root: ./scripts/docker-up.sh)
set -euo pipefail
_me="${BASH_SOURCE[0]:-}"
if [[ -n "$_me" && -e "$_me" ]]; then
  ROOT="$(cd "$(dirname "$_me")/.." && pwd)"
else
  ROOT="$(pwd)"
fi
unset _me
cd "$ROOT"

if [[ ! -f backend/.env ]]; then
  echo "ERROR: backend/.env missing. Create it before running Docker."
  exit 1
fi

echo "Building images…"
docker compose build

echo "Starting services…"
docker compose up -d

echo ""
echo "Done. Check: docker compose ps"
echo "Logs:    docker compose logs -f"
echo "Open in browser (use your EC2 public IP or domain):"
echo "  http://YOUR_PUBLIC_IP/"
echo ""
echo "Ensure backend/.env has FRONTEND_URL and APP_URL matching that URL (e.g. http://13.210.144.218)."
DUPSH

chmod +x ~/Audeint-AI/Audient-AI/scripts/docker-up.sh

NOTES — reading your logs
================================================================================
- Lines like "FDOCKER" / "NGXCONF" at end of a paste: that is the heredoc END marker —
  normal; next prompt means the file was written.

- "chmod: cannot access ... docker-up.sh: No such file or directory":
  scripts/ folder did not exist. BLOCK H now runs mkdir first; or run:
    mkdir -p ~/Audeint-AI/Audient-AI/scripts
  then re-paste BLOCK H.

- "Docker Compose is configured to build using Bake, but buildx isn't installed":
  harmless warning; build still succeeded. Optional fix:
    sudo apt-get update && sudo apt-get install -y docker-buildx-plugin

- Backend "Werkzeug ... development server": socketio.run() uses this; acceptable for
  FYP demo. Hardening later = eventlet/gevent + gunicorn (bigger change).

- Redis "vm.overcommit_memory": host tuning only. Optional on the EC2 host (not in container):
    sudo sysctl vm.overcommit_memory=1
  To persist: echo 'vm.overcommit_memory = 1' | sudo tee -a /etc/sysctl.conf

- Celery "running with superuser privileges": container runs as root (default image).
  Informational; optional later: non-root USER in Dockerfile.

- "Groq API : OK configured[Diarize]": two log lines printed close together — not an error.

- Stack healthy if: docker compose ps shows all Up, web maps 0.0.0.0:80->80, browser opens
  http://YOUR_PUBLIC_IP/
